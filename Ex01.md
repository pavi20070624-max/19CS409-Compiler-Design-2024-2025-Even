## AIM   
To write a C program to implement a symbol table.

## ALGORITHM
1.	Start the program.
2.	Get the input from the user with the terminating symbol ‘$’.
3.	Allocate memory for the variable by dynamic memory allocation function.
4.	If the next character of the symbol is an operator then only the memory is allocated.
5.	While reading, the input symbol is inserted into symbol table along with its memory address.
6.	The steps are repeated till ‘$’ is reached.
7.	To reach a variable, enter the variable to be searched and symbol table has been checked for corresponding variable, the variable along with its address is displayed as result.
8.	Stop the program. 

## PROGRAM
```
#include <stdio.h>
#include <ctype.h>
#include <string.h>
#include <stdlib.h>

#define MAX_EXPRESSION_SIZE 100
#define MAX_SYMBOLS 15

int main()
{
    int i = 0, j = 0, x = 0, n;
    void *add[MAX_SYMBOLS];

    char b[MAX_EXPRESSION_SIZE];
    char d[MAX_SYMBOLS];
    char c;

    printf("Enter the Expression terminated by $: ");

    while ((c = getchar()) != '$' && i < MAX_EXPRESSION_SIZE - 1)
    {
        b[i++] = c;
    }

    b[i] = '\0';

    n = i - 1;

    printf("\nGiven Expression: %s\n", b);

    printf("\nSymbol Table\n");
    printf("Symbol\taddr\ttype\n");

    for (j = 0; j <= n; j++)
    {
        c = b[j];

        if (isalpha((unsigned char)c))
        {
            /*
             * Check whether the character is an identifier.
             * It is considered an identifier if it is followed
             * by an operator or if it is the last character.
             */
            if (j == n)
            {
                void *p = malloc(sizeof(char));

                if (p == NULL)
                {
                    printf("Memory allocation failed.\n");
                    return 1;
                }

                add[x] = p;
                d[x] = c;

                printf("%c\t%p\tidentifier\n", c, p);

                x++;
            }
            else
            {
                char ch = b[j + 1];

                if (ch == '+' || ch == '-' ||
                    ch == '*' || ch == '=')
                {
                    void *p = malloc(sizeof(char));

                    if (p == NULL)
                    {
                        printf("Memory allocation failed.\n");
                        return 1;
                    }

                    add[x] = p;
                    d[x] = c;

                    printf("%c\t%p\tidentifier\n", c, p);

                    x++;
                }
            }
        }
    }

    /* Free allocated memory */
    for (i = 0; i < x; i++)
    {
        free(add[i]);
    }

    return 0;
}
```
